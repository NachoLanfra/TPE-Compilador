{{ TEMA 20 - ORDEN DE EVALUACION DE PARAMETROS REALES (OPCIONAL)
   f(...) [n1, n2, ...] indica el orden en que se copian los reales a los formales.
   Si no hay lista, se usa el orden de aparicion. La lista debe tener un valor por
   parametro y ser una permutacion de 1..n.
   Errores esperados en la lista final: 9. }}
prog_orden

ushortint a, b, c;
doublef d;

ushortint function f(ushortint p1, ushortint p2, ushortint p3)
begin
    ret(p1 + p2 + p3);
end;

doublef function fd(doublef q, ushortint r)
begin
    ret(q);
end;

begin
    a := 1$us;
    b := 2$us;
    {{ validos: con orden 1,2,3 y con orden 1,3,2 }}
    c := f(a, b = (4$us), a + b) [1$us, 2$us, 3$us];
    c := f(a, b = (4$us), a + b) [1$us, 3$us, 2$us];
    {{ valido: sin lista, se usa el orden de aparicion }}
    c := f(a, b, a);
    {{ valido: el primer real ushortint se convierte a doublef; orden 2,1 }}
    d := fd(a, b) [2$us, 1$us];
    {{ ERROR SEMANTICO: cantidad de parametros reales distinta a la de formales }}
    c := f(a, b);
    {{ ERROR SEMANTICO: la lista de orden tiene 2 valores y la funcion 3 parametros }}
    c := f(a, b, a) [1$us, 2$us];
    {{ ERROR SEMANTICO: la lista no es una permutacion (valor repetido) }}
    c := f(a, b, a) [1$us, 1$us, 2$us];
    {{ ERROR SEMANTICO: la lista no es una permutacion (valor fuera de rango) }}
    c := f(a, b, a) [1$us, 2$us, 9$us];
    {{ ERROR SEMANTICO: parametro doublef pasado a un formal ushortint }}
    c := f(a, d, a);
    {{ ERROR SEMANTICO: funcion no declarada }}
    c := fantasma(a);
    {{ ERROR SEMANTICO: resultado doublef asignado a un ushortint }}
    c := fd(d, a);
    {{ ERROR SINTACTICO: lista de orden mal formada }}
    c := f(a, b, a) [1$us, ];
end;
